```powershell
Set-Location -LiteralPath 'V:\_worktrees\Playbook\t3-51b04478'

$env:DOTNET_ADD_GLOBAL_TOOLS_TO_PATH = '0'
& 'C:\Program Files\dotnet\dotnet.exe' build ./src/PinguApps.Playbook -c Release
$server = (Resolve-Path ./src/PinguApps.Playbook/bin/Release/net10.0/PinguApps.Playbook.dll).Path

$empty = & 'C:\Program Files\nodejs\npx.ps1' @inspector --method tools/call --tool-name list_recipes --tool-arg targetVersion=net10.0 --tool-arg 'areas=["csharp"]' --tool-arg 'languages=["csharp"]' --tool-arg 'query=tangerine warehouse property normalization' --tool-arg includeShared=false --tool-arg limit=1 -- 'C:\Program Files\dotnet\dotnet.exe' $server mcp | ConvertFrom-Json
```
