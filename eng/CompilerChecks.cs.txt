using CatalogueCompiler;
using Microsoft.AspNetCore.Razor.Language;
using Microsoft.AspNetCore.Razor.Language.CodeGeneration;
using System.Reflection;
using System.Runtime.Loader;

var engine = RazorProjectEngine.Create(RazorConfiguration.Default, RazorProjectFileSystem.Create("."));
foreach (var input in new[] { "<h2>Catalogue: @DateTime.Now.Year</h2>", "<h2>@(" })
{
    var source = RazorSourceDocument.Create(input, "Catalogue.cshtml");
    var document = engine.Process(source, RazorFileKind.Legacy, [], TagHelperCollection.Empty, default);
#if PREVIOUS
    // Verification-only observation of the actual older compiler. Never teach this as supported API.
    var reader = typeof(RazorCodeDocument).GetMethod("GetCSharpDocument", BindingFlags.Instance | BindingFlags.NonPublic)!;
    Require(typeof(RazorCodeDocument).GetMethod("GetCSharpDocument") is null, "Older public-reader availability changed");
    var generated = (RazorCSharpDocument)reader.Invoke(document, null)!;
#else
    var generated = CatalogueTool.Compile(input);
#endif
    var copy = CatalogueTool.Copy(document, generated);
    Require(CatalogueTool.Read(copy).Contains("Catalogue", StringComparison.Ordinal), "Generated text missing");
    Require(copy.Text.ContentEquals(generated.Text), "Generated text changed");
    Require(copy.Diagnostics.SequenceEqual(generated.Diagnostics), "Diagnostics lost");
    Require(copy.SourceMappingsSortedByGenerated.SequenceEqual(generated.SourceMappingsSortedByGenerated), "Mappings lost");
    Require(copy.SourceMappingsSortedByOriginal.SequenceEqual(generated.SourceMappingsSortedByOriginal) && copy.LinePragmas.SequenceEqual(generated.LinePragmas), "Original-order mappings or line pragmas lost");
    Require(copy.SourceMappingsSortedByGenerated.Length > 0, "Actual source mappings missing");
    foreach (var mapping in copy.SourceMappingsSortedByGenerated)
    {
        Require(mapping.OriginalSpan.AbsoluteIndex >= 0 && mapping.OriginalSpan.Length >= 0 &&
            mapping.OriginalSpan.AbsoluteIndex <= input.Length && mapping.OriginalSpan.Length <= input.Length - mapping.OriginalSpan.AbsoluteIndex, "Original mapping outside source");
        Require(mapping.GeneratedSpan.AbsoluteIndex >= 0 && mapping.GeneratedSpan.Length >= 0 &&
            mapping.GeneratedSpan.AbsoluteIndex <= copy.Text.Length && mapping.GeneratedSpan.Length <= copy.Text.Length - mapping.GeneratedSpan.AbsoluteIndex, "Generated mapping outside output");
        var original = input.Substring(mapping.OriginalSpan.AbsoluteIndex, mapping.OriginalSpan.Length);
        var emitted = copy.Text.ToString().Substring(mapping.GeneratedSpan.AbsoluteIndex, mapping.GeneratedSpan.Length);
        Console.WriteLine($"Actual mapping retained: {mapping.OriginalSpan} -> {mapping.GeneratedSpan}; original={original}; generated-span={emitted}");
    }
    Require((copy.Diagnostics.Length > 0) == (input == "<h2>@("), "Invalid-input diagnostic mismatch");
#if !PREVIOUS
    Require(!copy.IsDeclarationDocument && !copy.IsStubDocument, "Normal output mislabeled");
    var labeled = CatalogueTool.Copy(document, generated, declaration: true, stub: true);
    Require(labeled.IsDeclarationDocument && labeled.IsStubDocument && labeled.Text.ContentEquals(copy.Text), "Flags must label, not regenerate, text");
    // Recompile the old five-argument source call against the actual new dependency set.
    var rebuilt = new RazorCSharpDocument(document, generated.Text, generated.Diagnostics, generated.SourceMappingsSortedByGenerated, generated.LinePragmas);
    Require(!rebuilt.IsDeclarationDocument && !rebuilt.IsStubDocument, "Optional defaults changed");
    if (generated.Diagnostics.Length > 0)
    {
        Require(CatalogueTool.DiagnosticCounts(source, generated.Diagnostics[0], true) == (1, 2), "Reporting context lost tree/explicit diagnostic");
        Require(CatalogueTool.DiagnosticCounts(source, generated.Diagnostics[0], false) == (0, 1), "Disabled tree import must still retain an explicit AddDiagnostic");
        var node = new Microsoft.AspNetCore.Razor.Language.Intermediate.DocumentIntermediateNode();
        node.AddDiagnostic(generated.Diagnostics[0]);
        using var defaults = new CodeRenderingContext(new CompilerProof.DiagnosticWriter(), source, node, RazorCodeGenerationOptions.Default);
        Require(defaults.GetDiagnostics().Length == 1, "Omitted reportDiagnostics must import the diagnostic");
        Console.WriteLine("PASS: omitted reportDiagnostics imports the diagnostic by default.");
        Console.WriteLine("PASS: reportDiagnostics true imports tree diagnostics; false skips tree import; explicit AddDiagnostic is retained in both.");
    }
#endif
    Console.WriteLine($"PASS: input={input}; generated={copy.Text.Length}; diagnostics={copy.Diagnostics.Length}; mappings={copy.SourceMappingsSortedByGenerated.Length}; actual mapping metadata retained");
}
#if !PREVIOUS
var previous = AssemblyLoadContext.Default.LoadFromAssemblyPath(Path.GetFullPath(args[0]));
var method = previous.GetType("CatalogueCompiler.CatalogueTool", throwOnError: true)!.GetMethod("Copy")!;
var abiDocument = engine.Process(RazorSourceDocument.Create("Catalogue", "Catalogue.cshtml"), RazorFileKind.Legacy, [], TagHelperCollection.Empty, default);
try { method.Invoke(null, [abiDocument, abiDocument.GetCSharpDocument()]); throw new InvalidOperationException("Expected old constructor ABI failure"); }
catch (TargetInvocationException error) when (error.InnerException is MissingMethodException)
{
    Console.WriteLine("PASS: actual previous compiled constructor consumer throws MissingMethodException; source defaults do not preserve ABI.");
}
#endif
foreach (var type in new[] { typeof(RazorCSharpDocument), typeof(CodeRenderingContext) })
    foreach (var constructor in type.GetConstructors()) Console.WriteLine("Actual contract: " + constructor);
Console.WriteLine("Sample runtime: " + System.Runtime.InteropServices.RuntimeInformation.FrameworkDescription);
static void Require(bool value, string message) { if (!value) throw new InvalidOperationException(message); }
