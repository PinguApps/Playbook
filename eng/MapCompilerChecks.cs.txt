static void CheckCopies(MapNode original, MapNode copy)
{
    Require(!ReferenceEquals(original, copy), "Root shared");
    Require(!ReferenceEquals(original.Colours, copy.Colours) && !ReferenceEquals(original.Notes, copy.Notes), "Mutable custom container shared");
    Require(ReferenceEquals(original.Notes["survey"], copy.Notes["survey"]), "Immutable note should be shared");
    CompareChildren(original, copy);
}
static void CompareChildren(IntermediateNode original, IntermediateNode copy)
{
    Require(!ReferenceEquals(original.Children, copy.Children) || original.Children.Count == 0, "Writable children shared");
    Require(original.Children.Count == copy.Children.Count, "Child count lost");
    for (var index = 0; index < original.Children.Count; index++)
    {
        var left = original.Children[index]; var right = copy.Children[index];
        Require(!ReferenceEquals(left, right) && left.Source == right.Source && left.IsImported == right.IsImported, "Child shared or common state lost");
        CompareChildren(left, right);
    }
}
static string Text(IntermediateNode node) => node is HtmlIntermediateToken token ? token.Content : string.Concat(node.Children.Select(Text));
static void CheckOriginalDiagnostic(IntermediateNode node) => Require(node.Diagnostics.Length == 1, "Diagnostic accumulation shared");
static string ExecuteGenerated(string source)
{
    var paths = ((string)AppContext.GetData("TRUSTED_PLATFORM_ASSEMBLIES")!).Split(Path.PathSeparator).Append(typeof(MapTemplate).Assembly.Location).Distinct();
    var compilation = CSharpCompilation.Create("GeneratedMap" + Guid.NewGuid().ToString("N"), [CSharpSyntaxTree.ParseText(source)],
        paths.Select(path => MetadataReference.CreateFromFile(path)), new CSharpCompilationOptions(OutputKind.DynamicallyLinkedLibrary));
    using var bytes = new MemoryStream();
    var result = compilation.Emit(bytes);
    Require(result.Success, "Generated C# compilation failed: " + string.Join(";", result.Diagnostics));
    bytes.Position = 0;
    var assembly = AssemblyLoadContext.Default.LoadFromStream(bytes);
    var type = assembly.GetTypes().Single(type => type.IsSubclassOf(typeof(MapTemplate)));
    var map = (MapTemplate)Activator.CreateInstance(type)!;
    map.ExecuteAsync().GetAwaiter().GetResult();
    return map.ToString();
}
static void Require(bool value, string message) { if (!value) throw new InvalidOperationException(message); }
