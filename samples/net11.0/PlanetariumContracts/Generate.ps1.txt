[CmdletBinding()]
param([ValidateSet('normal','duplicate','missing','explicit','empty','null-name')][string]$Case = 'normal')
$ErrorActionPreference = 'Stop'
$PSNativeCommandUseErrorActionPreference = $true
$dotnet = 'C:/Users/pingu/.dotnet-playbook-11/dotnet.exe'
$previousIntegration = $env:DOTNET_ADD_GLOBAL_TOOLS_TO_PATH
$previousCase = $env:PLANETARIUM_CASE
$outputDirectory = Join-Path ([IO.Path]::GetTempPath()) ('planetarium-contracts-' + [guid]::NewGuid().ToString('N'))
Push-Location -LiteralPath $PSScriptRoot
try {
    $env:DOTNET_ADD_GLOBAL_TOOLS_TO_PATH = '0'
    $env:PLANETARIUM_CASE = $Case
    & $dotnet restore ./PlanetariumContracts.csproj
    & $dotnet build ./PlanetariumContracts.csproj -c Release -t:Rebuild `
        -p:RestoreLockedMode=true -p:OpenApiGenerateDocuments=true `
        '-p:OpenApiGenerateDocumentsOptions=--openapi-version OpenApi3_0' `
        "-p:OpenApiDocumentsDirectory=$outputDirectory"
    & $dotnet ./bin/Release/net11.0/PlanetariumContracts.dll
    Get-Content -LiteralPath (Join-Path $outputDirectory 'PlanetariumContracts_public.json')
    Get-Content -LiteralPath (Join-Path $outputDirectory 'PlanetariumContracts_staff.json')
}
finally {
    Pop-Location
    $env:PLANETARIUM_CASE = $previousCase
    $env:DOTNET_ADD_GLOBAL_TOOLS_TO_PATH = $previousIntegration
}
// endsnippet:complete
